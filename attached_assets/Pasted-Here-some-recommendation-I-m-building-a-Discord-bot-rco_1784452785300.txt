I'm building a Discord bot (/rcookie command) that refreshes a Roblox .ROBLOSECURITY cookie.

```typescript
import axios from 'axios';
import { wrapper } from 'axios-cookiejar-support';
import { CookieJar } from 'tough-cookie';

export interface RefreshResult {
  newCookie: string | null;
  status: string;
}

export async function refreshCookie(cookie: string): Promise<RefreshResult> {
  console.log('[DEBUG] Starting refreshCookie with cookie length:', cookie.length);

  const jar = new CookieJar();
  await jar.setCookie(`.ROBLOSECURITY=${cookie.trim()}`, 'https://auth.roblox.com');
  console.log('[DEBUG] Cookie set in jar');

  const client = wrapper(
    axios.create({
      baseURL: 'https://auth.roblox.com',
      jar,
      withCredentials: true,
      headers: {
        'Content-Type': 'application/json',
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
      },
    })
  );

  try {
    // STEP 1 – get CSRF token
    console.log('[DEBUG] Sending POST /v2/logout ...');
    const csrfResp = await client.post('/v2/logout');
    console.log('[DEBUG] /v2/logout status:', csrfResp.status);
    console.log('[DEBUG] /v2/logout headers:', csrfResp.headers);

    const csrf = csrfResp.headers['x-csrf-token'] as string | undefined;
    console.log('[DEBUG] CSRF token found?', !!csrf, csrf ? csrf.slice(0, 20) + '...' : 'none');

    if (!csrf) {
      return { newCookie: null, status: `CSRF_FAILED_${csrfResp.status}` };
    }

    // STEP 2 – re‑authenticate
    console.log('[DEBUG] Sending POST /v2/logoutfromallsessionsandreauthenticate ...');
    const reauthResp = await client.post(
      '/v2/logoutfromallsessionsandreauthenticate',
      {},
      { headers: { 'X-CSRF-TOKEN': csrf } }
    );
    console.log('[DEBUG] Reauth status:', reauthResp.status);
    console.log('[DEBUG] Reauth headers:', reauthResp.headers);

    if (reauthResp.status === 200) {
      // Extract from jar
      const cookies = await jar.getCookies('https://auth.roblox.com');
      console.log('[DEBUG] All cookies in jar:', cookies.map(c => `${c.key}=${c.value.slice(0, 10)}...`));
      const robloxCookie = cookies.find(c => c.key === '.ROBLOSECURITY');

      if (robloxCookie) {
        console.log('[DEBUG] Found new .ROBLOSECURITY in jar');
        return { newCookie: robloxCookie.value, status: 'SUCCESS' };
      }

      // Fallback: parse Set-Cookie headers
      const setCookie = reauthResp.headers['set-cookie'];
      console.log('[DEBUG] Raw Set-Cookie header(s):', setCookie);
      if (Array.isArray(setCookie)) {
        for (const raw of setCookie) {
          const match = raw.match(/\.ROBLOSECURITY=([^;]+)/);
          if (match) {
            console.log('[DEBUG] Extracted from Set-Cookie header');
            return { newCookie: match[1], status: 'SUCCESS' };
          }
        }
      }

      console.warn('[DEBUG] No .ROBLOSECURITY found in response');
      return { newCookie: null, status: 'SUCCESS_NO_COOKIE' };
    }

    console.warn('[DEBUG] Reauth failed with status:', reauthResp.status);
    return { newCookie: null, status: `ERROR_${reauthResp.status}` };
  } catch (err: any) {
    console.error('[DEBUG] Exception:', err.message, err.response?.data || '');
    return { newCookie: null, status: `EXCEPTION_${err.message}` };
  }
}
